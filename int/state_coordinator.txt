import numpy as np
from multiprocessing import Process, Barrier, Queue, shared_memory
import time

# --- CONFIGURATION ---
STATE_SIZE = 1000000  # e.g., 1 million complex numbers (~16MB)
NUM_WORKERS = 4
TICKS = 10

def state_coordinator(shm_name, shape, dtype, write_queue, barrier):
    """The 'Owner' of the world state. Processes all writes before releasing workers."""
    # Attach to the existing shared memory
    existing_shm = shared_memory.SharedMemory(name=shm_name)
    world_state = np.ndarray(shape, dtype=dtype, buffer=existing_shm.buf)

    for _ in range(TICKS):
        # PHASE 1: Wait for workers to finish their math and submit writes
        barrier.wait()

        # PHASE 2: Drain the queue. All writes happen HERE and nowhere else.
        while not write_queue.empty():
            update = write_queue.get()
            idx, value = update
            world_state[idx] = value
        
        # PHASE 3: Signal that the State is now consistent and ready for Reading
        barrier.wait()

    existing_shm.close()

def worker(w_id, shm_name, shape, dtype, write_queue, barrier):
    """The Parallel Simulator."""
    existing_shm = shared_memory.SharedMemory(name=shm_name)
    world_state = np.ndarray(shape, dtype=dtype, buffer=existing_shm.buf)

    for t in range(TICKS):
        # --- CALCULATION PHASE ---
        # Read from world_state (Safe because Coordinator is waiting at barrier)
        local_val = world_state[w_id] 
        
        # Simulate heavy quantum math
        new_val = local_val + (w_id * 0.1) + t
        
        # Send write request to the queue
        write_queue.put((w_id, new_val))

        # --- SYNC PHASE ---
        barrier.wait() # Signal: "I am done writing"
        barrier.wait() # Wait for: "Coordinator is done updating"
        
        # Now world_state is consistent for the next tick
        if w_id == 0:
            print(f"Tick {t} complete. Sample state[0]: {world_state[0]}")

    existing_shm.close()

if __name__ == "__main__":
    # 1. Initialize Shared Memory (Complex numbers for Quantum State)
    dtype = np.complex128
    shape = (STATE_SIZE,)
    size_bytes = np.dtype(dtype).itemsize * STATE_SIZE
    shm = shared_memory.SharedMemory(create=True, size=size_bytes)
    
    # 2. Setup communication tools
    # Barrier for Workers + 1 Coordinator
    sync_barrier = Barrier(NUM_WORKERS + 1)
    write_queue = Queue()

    # 3. Start Processes
    processes = []
    coord = Process(target=state_coordinator, args=(shm.name, shape, dtype, write_queue, sync_barrier))
    coord.start()
    processes.append(coord)

    for i in range(NUM_WORKERS):
        p = Process(target=worker, args=(i, shm.name, shape, dtype, write_queue, sync_barrier))
        p.start()
        processes.append(p)

    # 4. Cleanup
    for p in processes:
        p.join()

    shm.close()
    shm.unlink() # Free the memory back to the OS

import numpy as np
from multiprocessing import Process, Barrier, shared_memory
import time

# --- CONFIGURATION ---
STATE_SIZE = 1000000 
NUM_WORKERS = 4
TICKS = 10

def state_coordinator(world_shm_name, mail_shm_name, shape, dtype, barrier):
    # Attach to World State (The "Read" area)
    world_shm = shared_memory.SharedMemory(name=world_shm_name)
    world_state = np.ndarray(shape, dtype=dtype, buffer=world_shm.buf)
    
    # Attach to Mailboxes (The "Write" area)
    mail_shm = shared_memory.SharedMemory(name=mail_shm_name)
    # Each row is a mailbox for one worker
    mailboxes = np.ndarray((NUM_WORKERS,), dtype=dtype, buffer=mail_shm.buf)

    for _ in range(TICKS):
        # PHASE 1: Wait for workers to finish calculations
        barrier.wait()

        # PHASE 2: Collect "Mail" (Zero-lock update)
        # We simply copy the values from the mailbox into the world state
        for i in range(NUM_WORKERS):
            world_state[i] = mailboxes[i]
        
        # PHASE 3: Signal: "World is updated, safe to read"
        barrier.wait()

    world_shm.close()
    mail_shm.close()

def worker(w_id, world_shm_name, mail_shm_name, shape, dtype, barrier):
    world_shm = shared_memory.SharedMemory(name=world_shm_name)
    world_state = np.ndarray(shape, dtype=dtype, buffer=world_shm.buf)
    
    mail_shm = shared_memory.SharedMemory(name=mail_shm_name)
    mailboxes = np.ndarray((NUM_WORKERS,), dtype=dtype, buffer=mail_shm.buf)

    for t in range(TICKS):
        # --- CALCULATION ---
        # 1. Read from world_state (Consistent global state)
        current_val = world_state[w_id]
        
        # 2. Compute quantum logic
        # For a 16MB state, you'd likely access specific indices here
        new_val = current_val + (w_id * 0.5) 
        
        # 3. Write to PRIVATE mailbox (No locking needed!)
        mailboxes[w_id] = new_val

        # --- SYNCHRONIZATION ---
        barrier.wait() # Done writing to my mailbox
        barrier.wait() # Waiting for Coordinator to move mail to world_state
        
        if w_id == 0:
            print(f"Tick {t} synced. Value: {world_state[0]}")

    world_shm.close()
    mail_shm.close()

if __name__ == "__main__":
    dtype = np.complex128
    
    # Create World State Memory
    world_shm = shared_memory.SharedMemory(create=True, size=np.dtype(dtype).itemsize * STATE_SIZE)
    # Create Mailbox Memory (One slot per worker)
    mail_shm = shared_memory.SharedMemory(create=True, size=np.dtype(dtype).itemsize * NUM_WORKERS)
    
    sync_barrier = Barrier(NUM_WORKERS + 1)

    procs = []
    # Start Coordinator
    c = Process(target=state_coordinator, args=(world_shm.name, mail_shm.name, (STATE_SIZE,), dtype, sync_barrier))
    c.start()
    procs.append(c)

    # Start Workers
    for i in range(NUM_WORKERS):
        p = Process(target=worker, args=(i, world_shm.name, mail_shm.name, (STATE_SIZE,), dtype, sync_barrier))
        p.start()
        procs.append(p)

    for p in procs:
        p.join()

    # Final Cleanup
    world_shm.close(); world_shm.unlink()
    mail_shm.close(); mail_shm.unlink()
